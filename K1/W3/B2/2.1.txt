session_start(); 
?>

<nav>
    <div class="logo"><strong>SurveyApp</strong></div>
    <!-- Een simpele link naar de beheeromgeving -->
    <a href="login.php" class="btn-admin">Admin Login</a>
</nav>

<div class="container">
    <h1>Vragenlijst</h1>
    
    <!-- PHP LOGICA VOOR BERICHTEN:
         Hieronder checkt PHP: "Staat er een succes-bericht in het geheugen?"
         Zo ja, dan maakt hij een groen blokje (div) en laat de tekst zien. -->
         
    <?php if(isset($_SESSION['success'])): ?>
        <div class="message success">
            <?php 
                echo $_SESSION['success']; // 'echo' betekent: plak de tekst hier in de HTML.
                unset($_SESSION['success']); // 'unset' wist het bericht uit het geheugen, anders blijft het staan.
            ?>
        </div>
    <?php endif; ?>

    <!-- Hetzelfde gebeurt hier voor foutmeldingen (bijvoorbeeld als een veld leeg is). -->

    <?php if(isset($_SESSION['error'])): ?>
        <div class="message error">
            <?php 
                echo $_SESSION['error']; 
                unset($_SESSION['error']); 
            ?>
        </div>
    <?php endif; ?>

    <!-- FORMULIER LOGICA:
         'action="submit.php"' vertelt de browser: "Als de gebruiker op de knop drukt, 
         stuur al deze antwoorden dan naar het bestand submit.php".
         'method="POST"' zorgt ervoor dat de antwoorden onzichtbaar worden verzonden (veilig). -->


    <form action="submit.php" method="POST" id="surveyForm">
        
        <div class="form-group">
            <label for="fullname">Vraag 1: Wat is je naam?</label>
            
            <!-- Het 'name' attribuut (fullname) is voor PHP het allerbelangrijkste. 
                 Dit is de 'label' waaronder de naam in de database wordt opgeslagen. -->

            <input type="text" id="fullname" name="fullname" required placeholder="Bijv. Jan Janssen">
        </div>

        <div class="form-group">
            <label for="phone">Vraag 2: Wat is je telefoon nummer?</label>
            <input type="tel" id="phone" name="phone" required placeholder="0612345678">
        </div>

        <div class="form-group">
            <label for="department">Vraag 3: Op welke afdeling werk je?</label>
            <select id="department" name="department" required>

                <option value="Marketing">Marketing</option>
                <option value="Sales">Sales</option>

            </select>
        </div>

        <button type="submit">Verstuur Antwoorden</button>
    </form>

<script>

    // JAVASCRIPT VALIDATIE:
    // Dit is een 'digitale uitsmijter'. Voordat de data naar PHP wordt gestuurd, 
    // checkt dit scriptje snel of de naam niet te kort is. 
    // Dit scheelt de server werk en geeft de gebruiker sneller feedback.

    document.getElementById('surveyForm').addEventListener('submit', function(e) {
        const name = document.getElementById('fullname').value;

        if(name.length < 2) {
            alert("Vul a.u.b. een geldige naam in.");
            e.preventDefault(); // Dit stopt het verzenden ("Halt, je mag er nog niet door!")
        }
    });
</script>
